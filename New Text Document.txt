def mat(name, color, rough=0.7, bump=0.0, scale=40):
    m = bpy.data.materials.new(name); m.use_nodes = True
    b = m.node_tree.nodes["Principled BSDF"]
    b.inputs["Base Color"].default_value = (*color, 1); b.inputs["Roughness"].default_value = rough
    if bump:
        n = m.node_tree.nodes.new("ShaderNodeTexNoise"); n.inputs["Scale"].default_value = scale
        bp = m.node_tree.nodes.new("ShaderNodeBump"); bp.inputs["Strength"].default_value = bump
        m.node_tree.links.new(n.outputs["Fac"], bp.inputs["Height"])
        m.node_tree.links.new(bp.outputs["Normal"], b.inputs["Normal"])
    return m

def box(name, loc, size, m, coll):
    bpy.ops.mesh.primitive_cube_add(location=loc)
    o = bpy.context.active_object; o.name = name; o.scale = (size[0]/2, size[1]/2, size[2]/2)
    o.data.materials.append(m); coll.objects.link(o); bpy.context.scene.collection.objects.unlink(o); return o

def prim(kind, name, loc, scale, m, coll):
    getattr(bpy.ops.mesh, f"primitive_{kind}_add")(location=loc)
    o = bpy.context.active_object; o.name = name; o.scale = scale
    o.data.materials.append(m); coll.objects.link(o); bpy.context.scene.collection.objects.unlink(o); return o

def load(path, name, loc, rot_z=0, coll=None):
    ext = path.lower().split('.')[-1]
    before = set(bpy.data.objects)
    if ext == 'fbx': bpy.ops.import_scene.fbx(filepath=path)
    elif ext in ('glb','gltf'): bpy.ops.import_scene.gltf(filepath=path)
    new = [o for o in bpy.data.objects if o not in before]
    root = bpy.data.objects.new(name, None); coll.objects.link(root)
    for o in new:
        if o.parent is None: o.parent = root
        for c in list(o.users_collection): c.objects.unlink(o)
        coll.objects.link(o)
    root.location = loc; root.rotation_euler[2] = math.radians(rot_z)
    return root
